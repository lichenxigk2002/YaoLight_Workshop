# 节点上下文传递设计方案（类似 Dify）

## 一、核心概念

### 1.1 节点输入输出字段定义
每个节点需要明确定义：
- **输入字段（Input Variables）**：节点需要从上游获取哪些数据
- **输出字段（Output Variables）**：节点会输出哪些数据供下游使用

### 1.2 字段映射机制
- 上游节点的输出字段 → 下游节点的输入字段
- 支持字段重命名和路径映射
- 支持多个上游节点的字段合并

### 2.2 上下文变量系统

```go
// VariableContext 变量上下文
// 存储所有已执行节点的输出变量，供下游节点引用
type VariableContext struct {
    // 格式：nodeId.variableName -> value
    // 例如：{"crawler_1.data": {...}, "crawler_1.text": "...", "dataclean_1.cleaned": [...]}
    variables map[string]interface{}
    // 节点输出数据（完整数据）
    nodeOutputs map[string]*NodeData
}

// NewVariableContext 创建变量上下文
func NewVariableContext() *VariableContext {
    return &VariableContext{
        variables:   make(map[string]interface{}),
        nodeOutputs: make(map[string]*NodeData),
    }
}

// SetNodeOutput 设置节点输出（同时提取变量）
func (vc *VariableContext) SetNodeOutput(nodeID string, nodeType string, data *NodeData) {
    vc.nodeOutputs[nodeID] = data
    
    // 获取节点输出定义
    def, exists := NodeIODefinitions[nodeType]
    if !exists {
        return
    }
    
    // 提取每个输出变量
    payload := ExtractPayload(data)
    for _, outputVar := range def.Outputs {
        var value interface{}
        
        // 从payload中提取变量值
        if payloadMap, ok := payload.(map[string]interface{}); ok {
            // 直接使用变量名作为key
            if val, exists := payloadMap[outputVar.Variable]; exists {
                value = val
            } else {
                // 尝试从data字段中提取
                if dataField, ok := payloadMap["data"]; ok {
                    if dataMap, ok := dataField.(map[string]interface{}); ok {
                        value = dataMap[outputVar.Variable]
                    }
                }
            }
        }
        
        // 存储变量：nodeId.variableName
        varKey := fmt.Sprintf("%s.%s", nodeID, outputVar.Variable)
        vc.variables[varKey] = value
    }
}

// GetVariable 获取变量值
// 支持路径：nodeId.variableName 或 variableName（自动查找）
func (vc *VariableContext) GetVariable(path string) (interface{}, error) {
    // 如果包含点号，说明是完整路径
    if strings.Contains(path, ".") {
        if value, exists := vc.variables[path]; exists {
            return value, nil
        }
    }
    
    // 否则在所有节点中查找该变量名
    for key, value := range vc.variables {
        parts := strings.Split(key, ".")
        if len(parts) == 2 && parts[1] == path {
            return value, nil
        }
    }
    
    return nil, fmt.Errorf("变量 %s 不存在", path)
}

// ResolveTemplate 解析模板字符串
// 支持 {{nodeId.variableName}} 或 {{variableName}} 语法
func (vc *VariableContext) ResolveTemplate(template string) (string, error) {
    // 使用正则表达式匹配 {{...}}
    re := regexp.MustCompile(`\{\{([^}]+)\}\}`)
    
    result := re.ReplaceAllStringFunc(template, func(match string) string {
        // 提取变量名
        varName := strings.Trim(match, "{{}}")
        varName = strings.TrimSpace(varName)
        
        // 获取变量值
        value, err := vc.GetVariable(varName)
        if err != nil {
            return match // 如果找不到，保持原样
        }
        
        // 转换为字符串
        return fmt.Sprintf("%v", value)
    })
    
    return result, nil
}
```

### 2.3 增强的爬虫节点实现

```go
func (e *WorkflowExecutor) executeCrawlerNode(node NodeConfig, dataFlow *DataFlow, ctx *VariableContext) NodeExecutionResult {
    // 获取上游数据
    inputData, _ := dataFlow.GetNodeInputData(node.ID)
    
    // 构建配置
    config := crawlerService.CrawlerConfig{
        URL: getString(node.Config, "url", ""),
        // ... 其他配置
    }
    
    // 如果URL包含模板变量，解析它
    if strings.Contains(config.URL, "{{") {
        resolvedURL, err := ctx.ResolveTemplate(config.URL)
        if err == nil {
            config.URL = resolvedURL
        }
    }
    
    // 如果URL为空，尝试从输入变量获取
    if config.URL == "" {
        if urlVar, err := ctx.GetVariable("url"); err == nil {
            config.URL = fmt.Sprintf("%v", urlVar)
        }
    }
    
    // 处理Headers中的变量
    if headersStr, ok := node.Config["headers"].(string); ok {
        if strings.Contains(headersStr, "{{") {
            resolvedHeaders, _ := ctx.ResolveTemplate(headersStr)
            // 解析JSON
            var headersMap map[string]interface{}
            if err := json.Unmarshal([]byte(resolvedHeaders), &headersMap); err == nil {
                for k, v := range headersMap {
                    config.Headers[k] = fmt.Sprintf("%v", v)
                }
            }
        }
    }
    
    // 执行爬虫
    result, err := e.crawlerService.Execute(config)
    // ... 处理结果
    
    // 输出标准化格式
    outputData := map[string]interface{}{
        "data": result.Data,  // 对应 output variable "data"
        "html": result.RawHTML, // 对应 output variable "html"
        "text": extractText(result.Data), // 对应 output variable "text"
    }
    
    return NodeExecutionResult{
        NodeID:  node.ID,
        Success: true,
        Data:    outputData,
    }
}
```

### 3.2 变量选择器组件

```typescript
// VariableSelector.tsx - 变量选择器组件
import React, { useState } from 'react';
import { Select, AutoComplete, Tooltip } from 'antd';
import { Node } from 'reactflow';

interface VariableSelectorProps {
  node: Node;
  nodes: Node[];
  edges: Edge[];
  value?: string;
  onChange?: (value: string) => void;
  placeholder?: string;
}

export const VariableSelector: React.FC<VariableSelectorProps> = ({
  node,
  nodes,
  edges,
  value,
  onChange,
  placeholder = '选择变量或输入表达式',
}) => {
  // 获取所有可用的上游变量
  const getAvailableVariables = () => {
    const variables: Array<{ label: string; value: string; nodeId: string }> = [];
    
    // 找到所有上游节点
    const upstreamNodes = edges
      .filter(edge => edge.target === node.id)
      .map(edge => nodes.find(n => n.id === edge.source))
      .filter(Boolean) as Node[];
    
    upstreamNodes.forEach(upstreamNode => {
      const nodeDef = nodeIODefinitions[upstreamNode.data?.type || upstreamNode.type];
      if (nodeDef?.outputs) {
        nodeDef.outputs.forEach(outputVar => {
          variables.push({
            label: `${upstreamNode.data?.label || upstreamNode.id}.${outputVar.label}`,
            value: `{{${upstreamNode.id}.${outputVar.variable}}}`,
            nodeId: upstreamNode.id,
          });
        });
      }
    });
    
    return variables;
  };

  const variables = getAvailableVariables();

  return (
    <AutoComplete
      value={value}
      onChange={onChange}
      placeholder={placeholder}
      options={variables.map(v => ({
        label: (
          <div>
            <span style={{ fontWeight: 500 }}>{v.label}</span>
            <Tooltip title={`变量路径: ${v.value}`}>
              <span style={{ color: '#999', marginLeft: 8, fontSize: 12 }}>
                {v.value}
              </span>
            </Tooltip>
          </div>
        ),
        value: v.value,
      }))}
      filterOption={(inputValue, option) =>
        option?.value?.toLowerCase().includes(inputValue.toLowerCase()) || false
      }
      style={{ width: '100%' }}
    />
  );
};
```

## 四、使用示例

### 4.1 工作流示例

```
开始节点
  ↓
爬虫节点 (crawler_1)
  输入: url = "https://example.com"
  输出: data, html, text
  ↓
数据清洗节点 (dataclean_1)
  输入: data = {{crawler_1.data}}
  输出: cleaned
  ↓
AI节点 (ai_1)
  输入: content = {{dataclean_1.cleaned}}
  输出: result
```
